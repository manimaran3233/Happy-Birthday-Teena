        // ========== AUTH & MUSIC ==========
        const CORRECT_PASSWORD = "12345";

        // ========== PO THE PANDA LOGIC ==========
        const poWrapper = document.getElementById('po-wrapper');
        const poBubble = document.getElementById('po-bubble');
        let poState = 'idle'; // idle, walking, dancing, sleeping
        let poIdleTimer;
        let walkTimeout;
        
        // Initial setup for starting position tracking
        poWrapper.style.transform = `translateX(-50%) scale(var(--po-scale))`;
        // We will move Po using CSS `left` property smoothly
        poWrapper.style.left = '50%'; 

        const poMessages = ["Hi, I'm Po!", "Moni is beautiful!", "Nom nom 🎋", "Happy Birthday! 🎂", "Time for cake! 🎉"];

        function setPoState(newState) {
            poWrapper.className = `state-${newState}`;
            poState = newState;
            
            // Handle facing direction persistence manually if needed, but wrapper classes control animation
            if(newState === 'walking') poWrapper.classList.add('state-walking');
            if(newState === 'dancing') poWrapper.classList.add('state-dancing');
            if(newState === 'sleeping') poWrapper.classList.add('state-sleeping');
            if(newState === 'idle') poWrapper.classList.add('state-idle');
        }

        function poSpeak(msg, duration = 3000) {
            if (poState === 'sleeping') return;
            poBubble.textContent = msg;
            poBubble.classList.add('show');
            setTimeout(() => { if (poBubble.textContent === msg) poBubble.classList.remove('show'); }, duration);
        }

        function resetPoIdle() {
            if (poState === 'sleeping') {
                setPoState('idle');
                poSpeak("Oh! I'm awake! 😳");
            }
            clearTimeout(poIdleTimer);
            poIdleTimer = setTimeout(() => {
                if (poState === 'idle') {
                    setPoState('sleeping');
                    poBubble.classList.remove('show');
                }
            }, 10000); // Sleeps after 10s of inactivity
        }

        function startPoLogic() {
            resetPoIdle();
            ['mousemove', 'scroll', 'touchstart'].forEach(evt => window.addEventListener(evt, resetPoIdle, {passive: true}));
            
            // Random speaking
            setInterval(() => {
                if (poState === 'idle' && Math.random() > 0.5) {
                    poSpeak(poMessages[Math.floor(Math.random() * poMessages.length)]);
                }
            }, 6000);

            // Desktop walking interaction
            document.addEventListener('click', (e) => {
                // Don't walk if clicking on Po or a button
                if (e.target.closest('#po-wrapper') || e.target.tagName === 'BUTTON' || poState === 'dancing') return;
                
                const targetX = e.clientX;
                const currentRect = poWrapper.getBoundingClientRect();
                const currentX = currentRect.left + (currentRect.width / 2);
                const distance = Math.abs(targetX - currentX);
                
                if (distance < 30) return; // Too close

                clearTimeout(walkTimeout);
                
                // Direction
                if (targetX < currentX) poWrapper.classList.add('facing-left');
                else poWrapper.classList.remove('facing-left');

                setPoState('walking');
                
                // Calculate time based on distance (speed = pixels per second)
                const speed = 250; 
                const duration = distance / speed;
                
                poWrapper.style.transition = `left ${duration}s linear`;
                poWrapper.style.left = `${targetX}px`;

                walkTimeout = setTimeout(() => {
                    if (poState === 'walking') setPoState('idle');
                }, duration * 1000);
            });

            // Click Po to dance
            poWrapper.addEventListener('click', (e) => {
                e.stopPropagation();
                if (poState === 'sleeping') { resetPoIdle(); return; }
                
                clearTimeout(walkTimeout);
                // Stop moving where he currently is
                const currentRect = poWrapper.getBoundingClientRect();
                poWrapper.style.transition = 'none';
                poWrapper.style.left = `${currentRect.left + (currentRect.width/2)}px`;

                setPoState('dancing');
                poSpeak("Yay! 🎉", 2000);
                createHeartBurst(currentRect.left + (currentRect.width/2), currentRect.top);

                setTimeout(() => { if (poState === 'dancing') setPoState('idle'); }, 2000);
            });
        }

        // Shake detection for mobile to blow candle
        let lastX = null, lastY = null, lastZ = null;
        function handleShake(e) {
            if (isCakeInteracted || !document.getElementById('page1').classList.contains('active')) return;
            let acc = e.accelerationIncludingGravity;
            if (!acc) return;
            
            if (lastX !== null) {
                let delta = Math.abs(lastX - acc.x) + Math.abs(lastY - acc.y) + Math.abs(lastZ - acc.z);
                if (delta > 25) { // Shake sensitivity threshold
                    processCakeInteraction('blow');
                }
            }
            lastX = acc.x; lastY = acc.y; lastZ = acc.z;
        }

        // Letter typewriter
        function startLetterTypewriter() {
            const paras = document.querySelectorAll('.love-message p');
            if(sessionStorage.getItem('typed')==='true') { paras.forEach(p=>p.classList.add('in-view')); return; }
            const texts = Array.from(paras).map(p => p.textContent);
            paras.forEach(p => p.textContent = '');
            let pIdx = 0, cIdx = 0;
            function type() {
                if (pIdx >= texts.length) { sessionStorage.setItem('typed', 'true'); return; }
                if (cIdx === 0) paras[pIdx].classList.add('in-view');
                paras[pIdx].textContent += texts[pIdx].charAt(cIdx++);
                if (cIdx >= texts[pIdx].length) { pIdx++; cIdx = 0; setTimeout(type, 300); }
                else setTimeout(type, 30);
            }
            type();
        }

        // Scroll animations (Photos, Timeline)
        function initScrollObserver() {
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(e => { if (e.isIntersecting) { e.target.classList.add('in-view', 'visible'); observer.unobserve(e.target); } });
            }, { threshold: 0.2 });
            document.querySelectorAll('.scroll-animate, .photo-item').forEach(el => observer.observe(el));
        }
